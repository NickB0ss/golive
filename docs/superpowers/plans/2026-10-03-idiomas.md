## Restrições globais

- Idiomas: exatamente `'pt-BR'`, `'en'`, `'es'`. Preferências: `'auto' | 'pt-BR' | 'en' | 'es'`. Com
  `'auto'`, vale o primeiro idioma do sistema cujo prefixo seja `pt`, `en` ou `es`; se nenhum servir, `en`.
- O pt-BR é a referência: toda chave nasce em `pt-BR.js`, com o texto **exatamente** como estava no código.
- As três línguas precisam ter as mesmas chaves e os mesmos `{marcadores}`. Os testes travam isso.
- Tom das traduções: en em inglês americano; es neutro, tratando por **tú**, sem *vosotros* e sem
  regionalismos. Use os termos do glossário (Tarefa 6).
- Proibido montar frase concatenando pedaços traduzidos (`t('a') + nome + t('b')`). Cada frase é **uma**
  chave com marcadores.
- `data-i18n` só vai em elemento cujo conteúdo é **só texto**. Se o elemento tiver ícone (`<svg>`) e texto,
  embrulhe o texto num `<span data-i18n="...">`.
- Em `innerHTML` montado por template, um valor que vem da pessoa (nome, chat) continua passando por
  `escapeHtml`. `t()` não escapa nada.
- Ficam em português de propósito: `console.*`, `logger.*`, o relatório copiado em Diagnóstico e as notas de
  versão do GitHub.
- O resultado de `traduzirCodigo()` pode carregar valores vindos de outro PC: ele só entra no DOM por
  `textContent` ou por atributo (`title`, `aria-label`), **nunca** por `innerHTML` sem `escapeHtml`.
- "Só códigos na rede" vale para texto **do sistema e dos bancos** (recusas, textos padrão, perguntas,
  palavras). Nome, chat, palpite e o que a pessoa digitou continuam indo como ela escreveu.
- Risco aceito (revisão de 2026-10-03): `traduzirCodigo` não distingue um código de um texto livre que, por
  acaso, seja exatamente uma chave existente. Quem digitar `mesa.enquete.sim` numa opção verá "Sim"/"Sí".
  Isso não dá a ninguém poder além de digitar a própria frase, então não justifica mudar o formato do estado.
- Regras de `src/renderer/AGENTS.md` continuam valendo:
  - linhas com até 120 colunas;
  - comentários em português, no estilo do arquivo;
  - ids são API: `ids-contrato.json` muda junto com o código, no mesmo diff;
  - só tokens no CSS.
- Validação de cada tarefa: `npm test` e `npm run lint` verdes. `node --test` **não carrega o `app.js`**: as
  tarefas que mexem no boot exigem abrir o app de verdade (`npm start`).

- [ ] **Passo 4: implementar `src/renderer/i18n/index.js`**

```js
'use strict';

/*
 * Traducao do app (spec 2026-10-03-idiomas). Mesmo UMD do resto do
 * renderer: a janela carrega por <script> (depois dos tres dicionarios), o
 * processo principal e o servidor da sala carregam por require.
 *
 * O idioma da janela vem do preload (window.golive.idioma, sincrono) e e
 * aplicado aqui mesmo, ao carregar -- antes de qualquer outro script do app
 * montar texto.
 */

(function (root) {
  const IDIOMAS = Object.freeze(['pt-BR', 'en', 'es']);
  const BASE = 'pt-BR';
  // Forma de uma chave: 'area.nome', 'mesa.truco.naoESuaVez'. Texto livre
  // (nome de pessoa, categoria digitada) nunca casa com isto por ter espaco,
  // acento ou nao ter ponto.
  const FORMA_CHAVE = /^[a-z][A-Za-z0-9]*(\.[A-Za-z0-9]+)+$/;

  function dicionarios() {
    const g = root.GoLive && root.GoLive.i18nDicionarios;
    if (g && g[BASE]) return g;
    if (typeof module !== 'undefined' && typeof require === 'function') {
      return { 'pt-BR': require('./pt-BR'), en: require('./en'), es: require('./es') };
    }
    return { [BASE]: {} };
  }

  let ativo = BASE;
  const avisadas = new Set();

  function definirIdioma(codigo) {
    ativo = IDIOMAS.includes(codigo) ? codigo : BASE;
    return ativo;
  }
  function idiomaAtivo() { return ativo; }

  function bruto(chave) {
    const d = dicionarios();
    const proprio = d[ativo] && d[ativo][chave];
    if (proprio !== undefined) return proprio;
    return d[BASE] ? d[BASE][chave] : undefined;
  }

  function forma(valor, n) {
    if (typeof valor === 'string') return valor;
    if (!valor || typeof valor !== 'object') return undefined;
    const qual = new Intl.PluralRules(ativo).select(Number(n) || 0);
    return valor[qual] !== undefined ? valor[qual] : valor.other;
  }

  function preencher(texto, valores) {
    return texto.replace(/\{(\w+)\}/g, (inteiro, nome) =>
      (valores && valores[nome] !== undefined && valores[nome] !== null ? String(valores[nome]) : inteiro));
  }

  function t(chave, valores) {
    const texto = forma(bruto(chave), valores && valores.n);
    if (typeof texto !== 'string') {
      if (!avisadas.has(chave)) {
        avisadas.add(chave);
        if (typeof console !== 'undefined') console.warn(`i18n: chave sem texto: ${chave}`);
      }
      return chave;
    }
    return preencher(texto, valores);
  }

  function existe(chave) {
    const d = dicionarios();
    return Boolean(d[BASE] && Object.prototype.hasOwnProperty.call(d[BASE], chave));
  }

  /** Chave + valores numa string so -- e o que atravessa a rede (o servidor
   * da sala repassa `detail` como string de ate 120 caracteres). */
  function codigo(chave, valores) {
    if (!valores) return chave;
    const q = new URLSearchParams();
    for (const [k, v] of Object.entries(valores)) q.set(k, String(v));
    const s = q.toString();
    return s ? `${chave}?${s}` : chave;
  }

  /** O inverso de `codigo`. Qualquer coisa que nao seja um codigo conhecido
   * (texto que a pessoa escreveu, por exemplo) volta como veio. */
  function traduzirCodigo(texto) {
    if (typeof texto !== 'string') return '';
    const corte = texto.indexOf('?');
    const chave = corte < 0 ? texto : texto.slice(0, corte);
    if (!FORMA_CHAVE.test(chave) || !existe(chave)) return texto;
    const valores = corte < 0 ? undefined : Object.fromEntries(new URLSearchParams(texto.slice(corte + 1)));
    return t(chave, valores);
  }

  function formatarHora(ts, opcoes) {
    const segundos = opcoes && opcoes.segundos ? { second: '2-digit' } : {};
    return new Date(ts).toLocaleTimeString(ativo, { hour: '2-digit', minute: '2-digit', ...segundos });
  }
  function formatarData(ts) {
    return new Date(ts).toLocaleDateString(ativo, { day: 'numeric', month: 'long' });
  }
  function maiuscula(s) {
    return s ? s.charAt(0).toLocaleUpperCase(ativo) + s.slice(1) : s;
  }

  /** `data-i18n="chave"` vira textContent; `data-i18n-attr="aria-label:chave;title:outra"`
   * vira atributo. Sem raiz, aplica no documento inteiro e acerta o <html lang>. */
  function aplicarNoDom(raiz) {
    const base = raiz || root.document;
    if (!base) return;
    base.querySelectorAll('[data-i18n]').forEach((el) => { el.textContent = t(el.dataset.i18n); });
    base.querySelectorAll('[data-i18n-attr]').forEach((el) => {
      for (const par of el.dataset.i18nAttr.split(';')) {
        const [attr, chave] = par.split(':').map((s) => s.trim());
        if (attr && chave) el.setAttribute(attr, t(chave));
      }
    });
    if (!raiz && base.documentElement) base.documentElement.lang = ativo;
  }

  function chaves(idioma) {
    return Object.keys(dicionarios()[idioma] || {});
  }

  // Idioma desta janela, entregue pelo preload antes de qualquer script.
  const doPreload = (root.golive && root.golive.idioma)
    || (root.goliveSpy && root.goliveSpy.idioma)
    || (root.goliveOverlay && root.goliveOverlay.idioma);
  if (doPreload && doPreload.ativo) definirIdioma(doPreload.ativo);

  const api = {
    IDIOMAS, definirIdioma, idiomaAtivo, t, codigo, traduzirCodigo, existe,
    formatarHora, formatarData, maiuscula, aplicarNoDom, chaves,
  };
  root.GoLive = root.GoLive || {};
  root.GoLive.i18n = api;
  if (typeof module !== 'undefined') module.exports = api;
})(typeof window !== 'undefined' ? window : globalThis);
```

const TOKEN_CSS = /^[a-z0-9]+(?:[-_]{1,2}[a-z0-9]+)*$/;
const FORMA_CHAVE = /^[a-z][A-Za-z0-9]*(\.[A-Za-z0-9]+)+(\?.*)?$/;

// Textos que ficam iguais em toda lingua (marca, nome proprio) ou que nao
// sao para pessoa ler. Cada entrada precisa de um motivo.
const EXCECOES = new Set([
  'GoLive LAN', // marca
]);

function ehListaDeClasses(s) {
  const tokens = s.trim().split(/\s+/);
  return tokens.every((x) => TOKEN_CSS.test(x)) && tokens.some((x) => /[-_]/.test(x));
}

function ehTextoVisivel(fragmento) {
  const s = String(fragmento).trim();
  if (!s || !/[A-Za-zÀ-ÿ]/.test(s)) return false;
  if (EXCECOES.has(s) || FORMA_CHAVE.test(s) || ehListaDeClasses(s)) return false;
  return ACENTO.test(s) || DUAS_PALAVRAS.test(s);
}

/** Texto de HTML estatico sem data-i18n: no de texto direto e os quatro
 * atributos que a pessoa le sem o par correspondente em data-i18n-attr. */
function literaisDoHtml(html) {
  const out = [];
  const linhaDe = (pos) => html.slice(0, pos).split('\n').length;
  const semScripts = html.replace(/<(script|style)\b[\s\S]*?<\/\1>/g, (m) => m.replace(/[^\n]/g, ' '));
  for (const m of semScripts.matchAll(/<([a-zA-Z][\w-]*)([^>]*)>([^<]+)/g)) {
    if (/\bdata-i18n="/.test(m[2])) continue;
    if (ehTextoVisivel(m[3])) out.push({ texto: m[3].trim(), linha: linhaDe(m.index) });
  }
  for (const m of semScripts.matchAll(/<[a-zA-Z][\w-]*([^>]*)>/g)) {
    const traduzidos = (m[1].match(/data-i18n-attr="([^"]*)"/) || [, ''])[1]
      .split(';').map((p) => p.split(':')[0].trim());
    for (const a of m[1].matchAll(/\b(title|aria-label|placeholder|alt)="([^"]*)"/g)) {
      if (!traduzidos.includes(a[1]) && ehTextoVisivel(a[2])) out.push({ texto: a[2], linha: linhaDe(m.index) });
    }
  }
  return out;
}

/** Literais visiveis de um .js, sem console.* nem logger.* (log e para o
 * desenvolvedor, fica em portugues -- spec, secao 2.4). */
function textosSoltos(arquivo, conteudo) {
  if (arquivo.endsWith('.html')) return literaisDoHtml(conteudo);
  const linhasDeLog = new Set();
  conteudo.split('\n').forEach((l, i) => { if (/\b(console|logger)\.\w+\(/.test(l)) linhasDeLog.add(i + 1); });
  const out = [];
  for (const { texto, linha } of extrairLiterais(conteudo)) {
    if (linhasDeLog.has(linha)) continue;
    for (const f of fragmentosVisiveis(texto)) if (ehTextoVisivel(f)) out.push({ texto: f.trim(), linha });
  }
  return out;
}

// Idioma do app (spec 2026-10-03-idiomas, secao 2). Puro e testavel sem
// Electron: quem chama passa app.getPreferredSystemLanguages() e o caminho
// de userData/idioma.json. A preferencia mora aqui, e nao no config.js,
// porque a splash e o processo principal nao enxergam o localStorage da janela.

const IDIOMAS = ['pt-BR', 'en', 'es'];
const PREFERENCIAS = ['auto', ...IDIOMAS];
const POR_PREFIXO = { pt: 'pt-BR', en: 'en', es: 'es' };

function normalizarPreferencia(p) {
  return PREFERENCIAS.includes(p) ? p : 'auto';
}

function resolverIdioma(preferencia, idiomasDoSistema) {
  const p = normalizarPreferencia(preferencia);
  if (p !== 'auto') return p;
  for (const tag of Array.isArray(idiomasDoSistema) ? idiomasDoSistema : []) {
    const prefixo = String(tag || '').toLowerCase().split(/[-_]/)[0];
    if (POR_PREFIXO[prefixo]) return POR_PREFIXO[prefixo];
  }
  return 'en';
}

function lerPreferencia(arquivo, fsMod = require('fs')) {
  try {
    return normalizarPreferencia(JSON.parse(fsMod.readFileSync(arquivo, 'utf8')).preferencia);
  } catch {
    return 'auto';
  }
}

function gravarPreferencia(arquivo, preferencia, fsMod = require('fs')) {
  const p = normalizarPreferencia(preferencia);
  fsMod.writeFileSync(arquivo, JSON.stringify({ preferencia: p }));
  return p;
}

module.exports = { IDIOMAS, PREFERENCIAS, resolverIdioma, lerPreferencia, gravarPreferencia };

### Procedimento de migração (vale para as Tarefas 7 a 12)

Toda tarefa de migração repete este ciclo, um arquivo ou trecho de cada vez:

1. **Listar:** `node tools/i18n/contar.js <arquivo> --de <N> --ate <M>` mostra `linha: texto` de cada texto
   solto no trecho.
2. **Trocar no código:**
   - **Texto em JS:** `'Sair da sala'` vira `t('sala.sair')`. Texto com valor vira uma chave com marcador:
     `` `${nome} entrou` `` vira `t('sala.entrou', { nome })`.
   - **Plural:** troque o `n === 1 ? ... : ...` por uma chave `{ one, other }` e `t('x', { n })`.
   - **Template de `innerHTML`:** use `${t('...')}`. Valor que vem de pessoa continua com `escapeHtml`
     **antes** de entrar em `t()`.
   - **HTML estático:** `data-i18n="chave"` no elemento só de texto, ou embrulhe o texto num `<span>`.
     Atributos com `data-i18n-attr="title:chave;aria-label:chave2"`. Mantenha o texto pt-BR no HTML como
     fallback.
   - **Constante de módulo com texto** (tabela de rótulos): guarde a **chave**, não o texto, e chame `t()`
     na hora de usar. Assim os testes conseguem trocar de idioma.
   - **Data e hora:** `toLocaleTimeString('pt-BR', …)` vira `formatarHora(ts)`, e o mesmo vale para
     `formatarData`. `toLocaleUpperCase('pt-BR')` vira `maiuscula`.
3. **Escrever as chaves:** pt-BR com o texto exato do código; en e es traduzidos pelo glossário. Para não
   colidir, prefixe pela área:
   `lobby.`, `salas.`, `sala.`, `presenca.`, `chat.`, `compartilhar.`, `config.`, `diag.`, `aviso.`, `erro.`,
   `atualizacao.`, `rabisco.`, `reacao.`, `espiar.`, `mesa.vista.`, `mesa.catalogo.`, `mesa.<tipo>.`,
   `quiz.`, `desenha.`.
   O nome depois do prefixo é curto, em camelCase e português (`sala.passarLideranca`). O mesmo texto em dois
   lugares com **o mesmo sentido** usa a mesma chave. Se o sentido for outro, use outra chave.
4. **Validar:**

'use strict';

/*
 * Banco do Quiz (spec 2026-10-03-idiomas, secao 5): 8 temas x 60, cada
 * pergunta nas tres linguas, com a MESMA resposta certa. A sala sorteia ids;
 * cada pessoa monta o texto no proprio idioma (texto()).
 */

(function (root) {
  const TEMAS = Object.freeze(['games', 'anime', 'futebol', 'esportes', 'musica', 'filmes', 'ciencia', 'mundo']);
  const LINGUAS = { 'pt-BR': 'pt', en: 'en', es: 'es' };

  function carregar(tema) {
    const g = root.GoLive && root.GoLive.mesaQuizBanco && root.GoLive.mesaQuizBanco[tema];
    if (g) return g;
    if (typeof module !== 'undefined') return require(`./quiz-banco/${tema}`);
    return [];
  }

  const perguntas = Object.freeze(TEMAS.flatMap((tema) =>
    carregar(tema).map((q) => Object.freeze({ ...q, tema }))));
  const indice = new Map(perguntas.map((q) => [q.id, q]));

  function porId(id) { return indice.get(id) || null; }

  function texto(q, idioma) {
    const [pergunta, alternativas] = q[LINGUAS[idioma] || 'pt'];
    return { pergunta, alternativas };
  }

  function validarBanco(minimoPorTema = 60) {
    const ids = new Set();
    for (const tema of TEMAS) {
      const doTema = perguntas.filter((q) => q.tema === tema);
      if (doTema.length < minimoPorTema) throw new Error(`Tema ${tema}: ${doTema.length} de ${minimoPorTema}`);
    }
    for (const q of perguntas) {
      if (typeof q.id !== 'string' || !q.id.startsWith(`${q.tema}-`) || ids.has(q.id)) throw new Error(`id: ${q.id}`);
      ids.add(q.id);
      if (![1, 2, 3].includes(q.nivel)) throw new Error(`nivel: ${q.id}`);
      if (!Number.isInteger(q.certa) || q.certa < 0 || q.certa > 3) throw new Error(`certa: ${q.id}`);

**Critérios** (spec, seção 5.1). Cada pergunta é reprovada se falhar em qualquer um:
